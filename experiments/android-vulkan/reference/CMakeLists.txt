cmake_minimum_required(VERSION 3.24)
project(CSE_ANDROID_LINUX_REFERENCE LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(M3D_REQUIRE_VULKAN ON CACHE BOOL "" FORCE)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../v16" "${CMAKE_CURRENT_BINARY_DIR}/v16")
add_executable(cse_android_reference ../reference.cpp)
target_link_libraries(cse_android_reference PRIVATE moteur3d_v16_core Qt6::Core Qt6::Gui)
